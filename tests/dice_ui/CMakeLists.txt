cmake_minimum_required(VERSION 3.16)
project(dice_ui_test LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "" FORCE)
add_subdirectory("../../managed_components/lvgl__lvgl" lvgl)
add_executable(dice_ui_test render.c ../../main/dice_ui.c ../../main/dice_model.c
               ../../assets/fonts/dice_font_16.c)
target_include_directories(dice_ui_test PRIVATE ../../main)
target_link_libraries(dice_ui_test PRIVATE lvgl)
if(NOT MSVC)
    target_compile_options(dice_ui_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
